12static std::string wrapText(const std::string& text, std::size_t maxLineLen = 120) {
13 std::string out;
14 std::string line;
15 std::istringstream iss(text);
16 std::string word;
17
18 while (iss >> word) {
19 if (line.size() + word.size() + 1 > maxLineLen) {
20 out += line + '\n';
21 line = word;
22 } else {
23 if (!line.empty()) line += ' ';
24 line += word;
25 }
26 }
27
28 if (!line.empty()) out += line;
29 return out;
30}

522void App::processEvents() {
523 while (const std::optional<sf::Event> event = window.pollEvent()) {
524 if (event->is<sf::Event::Closed>()) {
525 window.close();
526 }
527

972 if (scrollDownButton) scrollDownButton->setHovered(scrollDownButton->contains(mousePos));
973 if (editorSaveButton) editorSaveButton->setHovered(editorSaveButton->contains(mousePos));
974 if (editorBackButton) editorBackButton->setHovered(editorBackButton->contains(mousePos));
975 if (state == ScreenState::Quiz) {
976 engine.tick();
977 if (engine.timeLeft() == 0 && !engine.finished()) {
978 if (!engine.nextQuestion()) {
979 state = ScreenState::Result;
980 saveCurrentResult();
981 } else {
982 refreshAnswerButtons();
983 }
984 }
985 }
986
987 if (state == ScreenState::MultiplayerQuiz && !multiplayerQuestions.empty()) {
988 int elapsed = static_cast<int>(multiplayerClock.getElapsedTime().asSeconds());
989 int limit = multiplayerQuestions[multiplayerQuestionIndex].timeLimitSec;
990 multiplayerTimeLeft = std::max(0, limit - elapsed);
991
992 if (multiplayerTimeLeft == 0) {
993 advanceMultiplayerTurn(false);
994 }
995 }
996
997 refreshTexts();
